Configuration

One TOML file describes a campaign. It holds passwords in plain text, so load_config refuses a file that grants any permission to group or others. Create it from the template:

install -m 600 config.example.toml config.toml

Unknown keys and values of the wrong type are rejected before any node is contacted, and the message names the key. Relative local paths are resolved against the directory of the configuration file, and a leading ~ expands to the home directory. [push].local_source_dir, [pull].local_destination_root, and the target keys name, host, user, password, and remote_dir are mandatory; every other key has a default.

remote_dir and output_subdir must not contain .. segments or the characters *, ?, [, ], which rsync would expand on the node. Two targets on the same host and port must not share a remote_dir or nest one inside the other, whatever the purge_scope: a project purge of one would remove the output of the other before it is harvested, and both would be deployed into one tree. Host names are compared without case, IP addresses by the value the C library reads, as ssh does. The comparison concerns hosts, not storage: for nodes that mount one filesystem, such as a shared /home on a cluster, configure one target for it or give each target its own remote_dir. A numeric host with a field in octal or hexadecimal notation (192.0.2.012, 0xc0.0.2.10) is refused, and so is one that the C library does not read as an address (192.168.1.256, 2001:db8:10).

Template

config.example.toml from the repository root lists every key with its constraints:

# SshDataBridge configuration template. Install as config.toml with mode 600 and edit.

[globals]
# OpenSSH ConnectTimeout
connect_timeout = 10  # integer > 0; units: s

# Host-key verification policy
strict_host_key_checking = "accept-new"  # one of: "accept-new" | "yes" | "no"

# rsync compression (-z)
compress = true

# rsync bandwidth limit per transfer
bandwidth_limit = 0  # integer >= 0; units: KiB/s; 0 = unlimited

# Server silence before ssh sends a keepalive request (ServerAliveInterval)
server_alive_interval = 15  # integer >= 0; units: s; 0 = ssh client configuration decides

# Unanswered keepalive requests before ssh disconnects (ServerAliveCountMax)
server_alive_count_max = 3  # integer >= 1

[push]
# Local source directory deployed to every target (mandatory)
local_source_dir = "~/projects/sim_batch_01"  # path relative to this config file or absolute; '~' expands

# rsync exclude patterns for push (the default list)
excludes = [
    ".git",
    ".github",
    ".vscode",
    "*.swp",
    "*~",
]

# rsync include patterns applied before the .gitignore filter: deploy paths the project ignores
includes = []  # for example ["/Manifest.toml"] when every node runs the same Julia version

# Apply the .gitignore rules of the source tree as rsync excludes (no '!' negations)
use_gitignore = true

# Refuse to push from a working tree with uncommitted changes below local_source_dir
require_clean_git = false

[pull]
# Local destination root: one subdirectory per target, plus the rsync logs (mandatory)
local_destination_root = "~/campaigns/sim_batch_01/harvest"  # path relative to this config file or absolute; '~' expands

# Output directory relative to remote_dir, for targets that do not set their own
output_subdir = "output"  # relative path without '..' segments or '*', '?', '[', ']'

# Harvest only files matching these patterns (empty = all files); excludes take precedence; not combinable with a purge
includes = []

# rsync exclude patterns for pull (the default list)
excludes = [
    "*.tmp",
    "*~",
]

# Handling of an existing local directory of a target
collision_strategy = "resume"  # one of: "resume" | "backup" | "abort"

# Purge the remote directory after a verified harvest (requires --yes on the CLI)
clean_remote_after_pull = false

# Remote directory removed by a purge
purge_scope = "project"  # one of: "project" | "output"

[[targets]]
# Unique label (mandatory); names the local harvest directory and the rsync logs
name = "Cluster-Node-01"  # starts with a letter or digit; letters, digits, '.', '_', '-'; at most 64 characters

# Hostname, IPv4 address in decimal fields, or IPv6 literal without brackets
host = "192.168.1.100"

# SSH daemon port
port = 22  # integer in [1, 65535]

# Remote user name
user = "scientist"  # letter or underscore, then letters, digits, '.', '_', '-'; at most 32 characters

# Authentication password
password = "example_password_1"  # non-empty; no control characters

# Base directory of the campaign on the node; no other target on the same host and port may use it or a directory inside it
remote_dir = "/home/scientist/campaigns/sim_batch_01"  # absolute path without '..' segments or '*', '?', '[', ']'

# Remote output subdirectory relative to remote_dir (optional, defaults to [pull].output_subdir)
output_subdir = "output"  # relative path without '..' segments or '*', '?', '[', ']'

# Host-key policy override (optional)
strict_host_key_checking = "accept-new"  # one of: "accept-new" | "yes" | "no"

[[targets]]
name = "Cluster-Node-02"
host = "192.168.1.101"
port = 2222
user = "engineer"
password = "example_password_2"
remote_dir = "/home/engineer/campaigns/sim_batch_01"
output_subdir = "results"
